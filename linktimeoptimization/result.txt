в директории проекта

LD
cmake -S . -B build-ld \
  -DCMAKE_BUILD_TYPE=Release

MOLD
cmake -S . -B build-mold \
  -DCMAKE_BUILD_TYPE=Release \
  -DCMAKE_EXE_LINKER_FLAGS="-fuse-ld=mold" \
  -DCMAKE_SHARED_LINKER_FLAGS="-fuse-ld=mold" \
  -DCMAKE_MODULE_LINKER_FLAGS="-fuse-ld=mold"

cmake --build build-ld   -j"$(nproc)"
cmake --build build-mold -j"$(nproc)"

найти где лежит файл и инфой о линкере
find build-ld   -path '*/mysqld.dir/link.txt'
find build-mold -path '*/mysqld.dir/link.txt'

сам путь:
build-ld/sql/CMakeFiles/mysqld.dir/link.txt
build-mold/sql/CMakeFiles/mysqld.dir/link.txt

проверка что сборка завершилась, там будет mysqld и другие бинарники 
ls -la build-ld/sql/CMakeFiles/mysqld.dir/

удалил исполняемый файл 
rm -f build-mold/runtime_output_directory/mysqld
в двух дирах
rm -f build-ld/runtime_output_directory/mysqld

запустил заново линковку 
/usr/bin/time -f "real=%e user=%U sys=%S" cmake --build build-ld --target mysqld
/usr/bin/time -f "real=%e user=%U sys=%S" cmake --build build-mold --target mysqld

проверил что был использован молд 
readelf -p .comment build-mold/runtime_output_directory/mysqld | grep -i mold
с отладочной информацией здесь можно проверить время полной сборки проекта потом

# Перейдите в корень проекта MySQL
cd ~/mysql-server

# Удалите старые конфигурации (чтобы начать с чистого листа)
rm -rf build-ld-debug build-mold-debug

# Конфигурация для обычного линкера
cmake -S . -B build-ld-debug \
  -DCMAKE_BUILD_TYPE=Debug

# Конфигурация для mold
cmake -S . -B build-mold-debug \
  -DCMAKE_BUILD_TYPE=Debug \
  -DCMAKE_EXE_LINKER_FLAGS="-fuse-ld=mold" \
  -DCMAKE_SHARED_LINKER_FLAGS="-fuse-ld=mold" \
  -DCMAKE_MODULE_LINKER_FLAGS="-fuse-ld=mold"

# Сборка с обычным линкером
cmake --build build-ld-debug -j$(nproc)
# Сборка с mold
cmake --build build-mold-debug -j$(nproc)

# Удаляем бинарники
rm -f build-ld-debug/runtime_output_directory/mysqld
rm -f build-mold-debug/runtime_output_directory/mysqld

# Замер для обычного линкера
echo "=== GNU ld (Debug) ==="
/usr/bin/time -f "real=%e user=%U sys=%S" cmake --build build-ld-debug --target mysqld

# Замер для mold
echo "=== mold (Debug) ==="
/usr/bin/time -f "real=%e user=%U sys=%S" cmake --build build-mold-debug --target mysqld

скрипт sbatch 
mg511s28@pine:~/mysql-server$ cat build_mysql_mold.job 
#!/bin/bash
#SBATCH --job-name=mysql_mold
#SBATCH --output=mysql_mold_%j.out
#SBATCH --error=mysql_mold_%j.err
#SBATCH --time=24:00:00
#SBATCH --partition=2288
#SBATCH --nodes=1
#SBATCH --ntasks-per-node=1
#SBATCH --cpus-per-task=32
#SBATCH --mem=128G

echo "=== Job started at $(date) ==="
echo "Job ID: $SLURM_JOB_ID"
echo "Node: $SLURM_JOB_NODELIST"

export PATH="$HOME/cmake/bin:$HOME/mold_install/bin:$PATH"

echo "=== Checking versions ==="
cmake --version
mold --version


cd ~/mysql-server

echo "=== mold (Debug) ==="
/usr/bin/time -f "real=%e user=%U sys=%S" \
    cmake --build build-mold-debug --target mysqld -j${SLURM_CPUS_PER_TASK}

echo "=== Job finished at $(date) ==="
